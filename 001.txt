<script>
document.addEventListener('DOMContentLoaded', () => {

  // 共通のセットアップ関数
  function setupCharacterFade(selector, imageList, intervalMs = 7000) {
    const img = document.querySelector(selector);
    if (!img) return;

    let currentIndex = 0;

    // 最初の画像をセット（念のため）
    img.src = imageList[currentIndex];

    function changeCharacter() {
      // フェードアウト
      img.style.opacity = 0;

      setTimeout(() => {
        // インデックスを進める（0→1→0→1…）
        currentIndex = (currentIndex + 1) % imageList.length;
        img.src = imageList[currentIndex];

        // フェードイン
        img.style.opacity = 1;
      }, 800); // ← CSSのtransitionと合わせる
    }

    setInterval(changeCharacter, intervalMs);
  }

  // ▼ キャラ1：001 ↔ 002 ↔ 001…
  setupCharacterFade(".char1", [
    "image/character001.png",
    "image/character002.png"
  ], 7000);

  // ▼ キャラ2：011 ↔ 012 ↔ 011…
  setupCharacterFade(".char2", [
    "image/character011.png",
    "image/character012.png"
  ], 7000);

    // ▼ キャラ3：021 ↔ 022 ↔ 021…
  setupCharacterFade(".char3", [
    "image/character021.png",
    "image/character022.png"
  ], 7000);

    // ▼ キャラ4：021 ↔ 022 ↔ 021…
  setupCharacterFade(".char4", [
    "image/character031.png",
    "image/character032.png"
  ], 7000);

    // ▼ キャラ5：021 ↔ 022 ↔ 021…
  setupCharacterFade(".char5", [
    "image/character041.png",
    "image/character042.png"
  ], 7000);

});
</script>
