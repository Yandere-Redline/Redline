<!-- ★ここに立ち絵切り替えスクリプトを移動 -->
<script>
document.addEventListener('DOMContentLoaded', () => {

  const charImg = document.querySelector(".main-character");

  const charList = [
    "character001.png",
    "character002.png"
  ];

  let currentIndex = 0;

  function changeCharacter() {

    // 1. フェードアウト
    charImg.style.opacity = 0;

    // 2. 画像切り替え（0.8秒後）
    setTimeout(() => {
      currentIndex = (currentIndex + 1) % charList.length;
      charImg.src = charList[currentIndex];

      // 3. フェードイン
      charImg.style.opacity = 1;

    }, 800); // ← CSSの transition と同じ時間
  }

  // 5秒ごとに切り替え
  setInterval(changeCharacter, 7000);

});
</script>


<script>
  // すべてのサムネを取得
  const thumbs = document.querySelectorAll(".expansion");
  const popupBg = document.getElementById("popup-bg");
  const popupImg = document.getElementById("popup-img");

  thumbs.forEach(img => {
    img.addEventListener("click", () => {
      popupImg.src = img.src;   // 同じ画像を拡大表示
      popupBg.style.display = "flex";
    });
  });

  // 背景クリックで閉じる
  popupBg.addEventListener("click", () => {
    popupBg.style.display = "none";
  });
</script>
